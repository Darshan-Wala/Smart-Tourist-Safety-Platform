import React, { createContext, useContext, useEffect, useState } from 'react';
import AuthService from '../services/authService';
import { cacheProfile, getCachedProfile, clearCachedProfile } from '../utils/profileCache';
import { isDigitalIDValid } from '../utils/digitalID';

const AuthContext = createContext(null);
export const useAuth = () => useContext(AuthContext);

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [profile, setProfile] = useState(null);
  const [loading, setLoading] = useState(true);

  // Quick hydrate from cache
  useEffect(() => {
    (async () => {
      const cached = await getCachedProfile();
      if (cached) setProfile(cached);
    })();
  }, []);

  // Auth listener
  useEffect(() => {
    const unsubscribe = AuthService.listenAuth(async (firebaseUser) => {
      if (firebaseUser) {
        setUser(firebaseUser);
        // We re-fetch profile via login or on demand; for listener-only, we can lazy load.
        // Optionally fetch from Firestore again if needed.
      } else {
        setUser(null);
        setProfile(null);
        clearCachedProfile();
      }
      setLoading(false);
    });
    return unsubscribe;
  }, []);

  const register = async (data) => {
    setLoading(true);
    try {
      const { user: u, profile: p } = await AuthService.register(data);
      setUser(u);
      setProfile(p);
      cacheProfile(p);
      return { user: u, profile: p };
    } finally {
      setLoading(false);
    }
  };

  const login = async (email, password) => {
    setLoading(true);
    try {
      const { user: u, profile: p } = await AuthService.login(email, password);
      setUser(u);
      setProfile(p);
      if (p) cacheProfile(p);
      return { user: u, profile: p };
    } finally {
      setLoading(false);
    }
  };

  const logout = async () => {
    setLoading(true);
    try {
      await AuthService.logout();
      setUser(null);
      setProfile(null);
      clearCachedProfile();
    } finally {
      setLoading(false);
    }
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        profile,
        loading,
        isAuthenticated: !!user,
        register,
        login,
        logout,
        isDigitalIDValid
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};
